cmake_minimum_required(VERSION 3.16)
project(bms_backend C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MICROHTTPD REQUIRED IMPORTED_TARGET libmicrohttpd)
pkg_check_modules(CJSON REQUIRED IMPORTED_TARGET libcjson)
add_library(bms_core STATIC src/bms.c)
target_include_directories(bms_core PUBLIC include)
target_link_libraries(bms_core PUBLIC Threads::Threads m)
target_compile_options(bms_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(bms-server src/server.c)
target_link_libraries(bms-server PRIVATE bms_core PkgConfig::MICROHTTPD PkgConfig::CJSON)
target_compile_options(bms-server PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
include(CTest)
if(BUILD_TESTING)
 add_executable(test_bms tests/test_bms.c)
 target_link_libraries(test_bms PRIVATE bms_core)
 add_test(NAME bms_core_tests COMMAND test_bms)
endif()
